cmake_minimum_required(VERSION 3.21)
project(omacut VERSION 0.7.4 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Gui Widgets Qml Quick QuickControls2 Multimedia)
set(OMACUT_SOURCES src/backend.cpp src/edit.cpp src/ffmpeg.cpp src/thumbprovider.cpp src/timeline.cpp
    src/backend.h src/edit.h src/ffmpeg.h src/thumbprovider.h src/timeline.h
    src/shortcutsettings.h src/clipmovekeys.h src/filepicker.h)
set(OMACUT_LIBS Qt6::Core Qt6::Gui Qt6::Widgets Qt6::Qml Qt6::Quick Qt6::QuickControls2 Qt6::Multimedia)
if(NOT WIN32)
    find_package(Qt6 REQUIRED COMPONENTS DBus)
    list(APPEND OMACUT_SOURCES src/portalfilepicker.cpp src/portalfilepicker.h)
    list(APPEND OMACUT_LIBS Qt6::DBus)
endif()
if(APPLE)
    enable_language(OBJCXX)
    list(APPEND OMACUT_SOURCES src/macfilepicker.mm src/macfilepicker.h)
    set_source_files_properties(src/macfilepicker.mm PROPERTIES COMPILE_OPTIONS -fobjc-arc)
    list(APPEND OMACUT_LIBS "-framework AppKit" "-framework UniformTypeIdentifiers")
else()
    list(APPEND OMACUT_SOURCES src/desktopfilepicker.cpp src/desktopfilepicker.h)
endif()
add_library(omacut_backend STATIC ${OMACUT_SOURCES})
target_include_directories(omacut_backend PUBLIC src)
target_link_libraries(omacut_backend PUBLIC ${OMACUT_LIBS})
add_executable(omacut src/main.cpp src/smoke.cpp src/smoke.h src/resources.qrc)
target_link_libraries(omacut PRIVATE omacut_backend)
if(WIN32)
    set_target_properties(omacut PROPERTIES WIN32_EXECUTABLE TRUE)
endif()
include(CTest)
if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    add_executable(backend_tests tests/backend_tests.cpp)
    target_link_libraries(backend_tests PRIVATE omacut_backend Qt6::Test)
    add_test(NAME backend COMMAND backend_tests)
    set_tests_properties(backend PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software")
endif()
install(TARGETS omacut RUNTIME DESTINATION bin)
install(FILES pkgbuild/omacut.desktop DESTINATION share/applications)
install(FILES pkgbuild/omacut.svg DESTINATION share/icons/hicolor/scalable/apps)
install(FILES LICENSE README.md DESTINATION share/doc/omacut)
